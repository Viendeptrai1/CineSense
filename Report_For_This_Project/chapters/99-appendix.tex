\chapter{Phụ lục}

\section{Danh sách module chính}
\begin{itemize}
    \item \texttt{api/}: backend FastAPI.
    \item \texttt{etl\_pipeline/}: crawler, schema và lớp chuẩn hóa dữ liệu.
    \item \texttt{training/}: huấn luyện, đánh giá và xuất artifacts recommendation.
    \item \texttt{frontend/}: giao diện catalog, chi tiết phim và recommendation.
    \item \texttt{scripts/}: audit, restore, backup, pipeline orchestration.
    \item \texttt{Report\_For\_This\_Project/}: toàn bộ nguồn LaTeX của báo cáo.
\end{itemize}

\section{Lệnh chạy hệ thống ở mức tối thiểu}
\begin{lstlisting}[language=bash,caption={Khởi động discovery runtime}]
docker-compose up -d
source .venv/bin/activate
uvicorn api.main:app --reload --port 8000
cd frontend
python3 -m http.server 3000
\end{lstlisting}

\section{Lệnh chạy pipeline recommendation}
\begin{lstlisting}[language=bash,caption={Chuỗi lệnh huấn luyện và đánh giá}]
python scripts/audit_core_data.py --output-json training/artifacts/latest/core_audit.json
python -m training.baselines.train_tfidf --artifact-name tfidf_latest --top-k 20
python -m training.models.train_sentence_transformer --artifact-name sbert_latest --top-k 20
python -m training.evaluation.run_eval --artifact-dir training/artifacts/sbert_latest --output-json training/artifacts/sbert_latest/eval.json
\end{lstlisting}

\section{Danh mục tài nguyên minh họa cần thu thập}
\begin{itemize}
    \item ảnh giao diện catalog ở nhiều trạng thái dữ liệu;
    \item ảnh trang chi tiết với các phim có/không có review;
    \item ảnh danh sách recommendation từ artifact baseline và improved;
    \item biểu đồ thống kê dữ liệu (độ dài text, phân bố review/phim);
    \item bảng kết quả định lượng từ \texttt{eval.json};
    \item video demo dự phòng dài 3--5 phút.
\end{itemize}

\section{Danh sách kiểm tra trước khi nộp}
\begin{itemize}
    \item thay toàn bộ giá trị ``N/A'' bằng số liệu thực nghiệm trước khi nộp;
    \item tất cả hình/bảng đều có caption, label và được dẫn chiếu trong nội dung;
    \item số liệu trong bảng trùng khớp với artifact JSON lưu trong dự án;
    \item tài liệu tham khảo được trích dẫn nhất quán trong toàn báo cáo;
    \item phần kết luận phản ánh đúng kết quả thực nghiệm, không suy diễn quá mức.
\end{itemize}
